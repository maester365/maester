  ```powershell
  function Test-MtEmergencyAccessPhishingResistantMethod {
      <#
      .SYNOPSIS
      Checks that every emergency access (break glass) account has a phishing-resistant authentication method registered.
  
      .DESCRIPTION
      Emergency access accounts are permanent Global Administrators that are excluded from Conditional Access, so
      the authentication method registered on the account is what protects it. Microsoft recommends a passkey
      (FIDO2) for these accounts: it cannot be phished or relayed by an adversary-in-the-middle proxy, and it
      satisfies the multifactor authentication that the Microsoft admin portals require.
  
      The accounts come from the EmergencyAccessAccounts setting. A configured group is expanded to its user
      members, nested groups included. For each account the test reads the registered authentication methods and
      passes when every account has at least one passkey (FIDO2), Windows Hello for Business or platform
      credential method. The test is skipped when no emergency access accounts are configured.
  
      Certificate-based authentication is not registered on the user object, so an account that signs in only
      with a certificate fails this test.
  
      .EXAMPLE
      Test-MtEmergencyAccessPhishingResistantMethod
  
      Returns true when every configured emergency access account has a phishing-resistant method registered.
  
      .LINK
      https://maester.dev/docs/tests/MT.1266
      #>
      [MaesterTest(
          Id                 = 'MT.1266',
          Title              = 'Emergency access accounts should have a phishing-resistant authentication method registered.',
          Product            = 'entra',
          Feature            = 'authentication-methods',
          Area               = 'privileged-access',
          ResourceType       = 'User',
          Severity           = 'High',
          UserImpact         = 'Low',
          ImplementationCost = 'Low',
          Mitre              = ('T1078.004', 'T1566', 'T1557'),
          Control            = ('nist-800-53-r5:IA-2(1)', 'nist-800-53-r5:IA-2(8)'),
          Service            = 'Graph',
          Permission         = ('Graph:Scope:UserAuthenticationMethod.Read.All', 'Graph:Scope:User.Read.All | Graph:Scope:Directory.Read.All', 'Graph:Scope:GroupMember.Read.All | Graph:Scope:Directory.Read.All'),
          Author             = 'merill'
      )]
      [CmdletBinding()]
      [OutputType([bool])]
      param(
          # Emergency access (break glass) accounts and groups, from the EmergencyAccessAccounts setting.
          [MaesterParameter(Setting = 'EmergencyAccessAccounts')]
          [object[]] $EmergencyAccessAccounts = @()
      )
  
      # The method types Microsoft Entra ID treats as phishing-resistant, with the name the result shows.
      $phishingResistantMethods = @{
          '#microsoft.graph.fido2AuthenticationMethod'                   = 'Passkey (FIDO2)'
          '#microsoft.graph.windowsHelloForBusinessAuthenticationMethod' = 'Windows Hello for Business'
          '#microsoft.graph.platformCredentialAuthenticationMethod'      = 'Platform credential'
      }
  
      # Throws when a configured account or group cannot be resolved, so the result is an error and not a pass.
      $configuredAccounts = @(Get-MtEmergencyAccessAccount -Account $EmergencyAccessAccounts | Where-Object { $_ })
      if ($configuredAccounts.Count -eq 0) {
          Add-MtTestResultDetail -SkippedBecause Custom -SkippedCustomReason 'No emergency access accounts are configured. Add your break glass accounts to the EmergencyAccessAccounts setting (GlobalSettings in maester-config.json) so this test can check them. See https://maester.dev/docs/configuration/emergency-access-accounts'
          return $null
      }
  
      # The user accounts to check: the configured users and the user members of the configured groups.
      $userNames = [ordered]@{}
      foreach ($account in $configuredAccounts) {
          if ($account.Type -eq 'group') {
              $members = @(Get-MtGroupMember -GroupId $account.ObjectId -Recursive | Where-Object { $_.'@odata.type' -eq '#microsoft.graph.user' })
              foreach ($member in $members) {
                  $userNames[[string]$member.id] = $member.displayName
              }
          } else {
              $userNames[[string]$account.ObjectId] = $account.DisplayName
          }
      }
  
      if ($userNames.Count -eq 0) {
          Add-MtTestResultDetail -SkippedBecause Custom -SkippedCustomReason 'The groups in the EmergencyAccessAccounts setting have no user members, so there is no emergency access account to check.'
          return $null
      }
  
      $users = foreach ($userId in $userNames.Keys) {
          $registered = @((Get-MtUserAuthenticationMethod -UserId $userId).AuthenticationMethods | Where-Object { $_ })
          $methodNames = @($registered | ForEach-Object { $phishingResistantMethods[[string]$_.'@odata.type'] } | Where-Object { $_ } | Sort-Object -Unique)
          [pscustomobject]@{
              id          = $userId
              displayName = if ($userNames[$userId]) { $userNames[$userId] } else { $userId }
              Methods     = $methodNames
          }
      }
      $users = @($users)
      $usersWithoutMethod = @($users | Where-Object { $_.Methods.Count -eq 0 })
  
      $portalLink = "$($__MtSession.AdminPortalUrl.Entra)#view/Microsoft_AAD_UsersAndTenants/UserProfileMenuBlade/~/UserAuthMethods/userId/{0}"
      $table = "| Emergency access account | Phishing-resistant methods | Status |`n| --- | --- | --- |`n"
      foreach ($user in $users) {
          $name = "[$(Get-MtSafeMarkdown $user.displayName)]($($portalLink -f $user.id))"
          if ($user.Methods.Count -gt 0) {
              $table += "| $name | $($user.Methods -join ', ') | ✅ Registered |`n"
          } else {
              $table += "| $name | None | ❌ Not registered |`n"
          }
      }
  
      if ($usersWithoutMethod.Count -eq 0) {
          Add-MtTestResultDetail -Result "Well done. Every emergency access account has a phishing-resistant authentication method registered.`n`n$table"
          return $true
      }
  
      $result = "These emergency access accounts have no phishing-resistant authentication method registered. Register a passkey (FIDO2) for each of them:`n`n%TestResult%`n`n$table"
      Add-MtTestResultDetail -Result $result -GraphObjects $usersWithoutMethod -GraphObjectType Users
      return $false
  }
  ```

  ### `Test.MT.1266.md` (final)
  
  ```markdown
  Every emergency access (break glass) account should have a phishing-resistant authentication method registered, such as a passkey on a FIDO2 security key.
  
  Emergency access accounts are permanent Global Administrators, and they are excluded from Conditional Access so that a policy mistake or an outage cannot lock you out of the tenant. That exclusion also means no policy protects their sign-in: the authentication method registered on the account is the control. If the account has only a password, or a method such as a text message or an authenticator code, an attacker who phishes the credentials or relays the sign-in through an adversary-in-the-middle proxy signs in as a Global Administrator. A passkey is bound to the real sign-in page, so it cannot be phished or relayed. It also satisfies the multifactor authentication that the Microsoft admin portals require, so the account still works in an emergency.
  
  Maester reads the accounts from the `EmergencyAccessAccounts` setting. A group in the setting is checked through its user members, including members of nested groups. The test passes when every account has at least one of these methods registered:
  
  - Passkey (FIDO2), on a security key or in Microsoft Authenticator
  - Windows Hello for Business
  - Platform credential for macOS
  
  The test is skipped when no emergency access accounts are configured. See [Emergency access accounts](https://maester.dev/docs/configuration/emergency-access-accounts) to tell Maester which accounts they are.
  
  The test does not see certificate-based authentication, because a certificate is not registered on the user. If your emergency access accounts sign in only with a certificate, turn this test off in your configuration (`"Enabled": false` in its `TestSettings` row). It also does not check that weaker methods are absent or that the method is enforced at sign-in.
  
  #### Remediation action
  
  Microsoft recommends a passkey (FIDO2) on a security key for emergency access accounts, different from the method your everyday administrator accounts use.
  
  1. In the [Microsoft Entra admin center](https://entra.microsoft.com), go to **Entra ID** > **Authentication methods** > **[Policies](https://entra.microsoft.com/#view/Microsoft_AAD_IAM/AuthenticationMethodsMenuBlade/~/AdminAuthMethods)**, open **Passkey (FIDO2)** and make sure it is enabled for the emergency access accounts.
  2. Sign in as each emergency access account listed in the result, open [Security info](https://mysignins.microsoft.com/security-info), select **Add sign-in method** > **Security key** and follow the steps to register the key.
  3. Confirm the key is listed under **Users** > the account > **Authentication methods**. The account names in the result link to that page.
  4. Store each key in a secure place that more than one authorized administrator can reach, such as a safe, and test the sign-in at least every 90 days.
  
  #### Related links
  
  * [Manage emergency access admin accounts in Microsoft Entra ID](https://learn.microsoft.com/entra/identity/role-based-access-control/security-emergency-access)
  * [Enable passkeys (FIDO2) for your organization](https://learn.microsoft.com/entra/identity/authentication/how-to-authentication-passkeys-fido2)
  * [Register a passkey using a FIDO2 security key](https://learn.microsoft.com/entra/identity/authentication/how-to-register-passkey-with-security-key)
  * [Mandatory multifactor authentication for Azure and admin portals](https://learn.microsoft.com/entra/identity/authentication/concept-mandatory-multifactor-authentication)
  ```
